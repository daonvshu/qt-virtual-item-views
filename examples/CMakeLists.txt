function(viv_add_example name)
    add_executable(${name} ${ARGN} common/demoscrolling.cpp)
    target_link_libraries(${name} PRIVATE VirtualItemViews Qt${QT_VERSION_MAJOR}::Widgets viv_demo_scrolling)
    set_property(DIRECTORY APPEND PROPERTY VIV_EXAMPLE_TARGETS ${name})
    if(MSVC)
        target_compile_options(${name} PRIVATE /utf-8)
    endif()
endfunction()

option(VIRTUALITEMVIEWS_EXAMPLES_USE_SMOOTH_SCROLLBAR "Use optional SmoothScrollbar in examples" ON)
add_library(viv_demo_scrolling INTERFACE)
if(VIRTUALITEMVIEWS_EXAMPLES_USE_SMOOTH_SCROLLBAR)
    # The package config discovers Qt dynamically, so inspect the exported
    # target's actual Qt dependency before importing a prebuilt package.
    find_file(_viv_smooth_config NAMES SmoothScrollbarConfig.cmake
        HINTS "${SmoothScrollbar_DIR}" ENV SmoothScrollbar_DIR ${CMAKE_PREFIX_PATH} ENV CMAKE_PREFIX_PATH
        PATH_SUFFIXES SmoothScrollbar lib/cmake/SmoothScrollbar)
    if(_viv_smooth_config)
        get_filename_component(_viv_smooth_dir "${_viv_smooth_config}" DIRECTORY)
        if(EXISTS "${_viv_smooth_dir}/SmoothScrollbarTargets.cmake")
            file(STRINGS "${_viv_smooth_dir}/SmoothScrollbarTargets.cmake" _viv_smooth_qt
                REGEX "INTERFACE_LINK_LIBRARIES.*Qt[56]::")
            if(_viv_smooth_qt MATCHES "Qt${QT_VERSION_MAJOR}::Widgets")
                find_package(SmoothScrollbar CONFIG QUIET PATHS "${_viv_smooth_dir}" NO_DEFAULT_PATH)
            else()
                message(STATUS "Examples: SmoothScrollbar Qt version does not match; using native scrolling")
            endif()
        endif()
    endif()
    unset(_viv_smooth_config CACHE)
    if(TARGET SmoothScrollbar::SmoothScrollbar)
        target_link_libraries(viv_demo_scrolling INTERFACE SmoothScrollbar::SmoothScrollbar)
        target_compile_definitions(viv_demo_scrolling INTERFACE VIV_DEMO_USE_SMOOTH_SCROLLBAR=1)
        set(_viv_demo_smooth_enabled TRUE)
    endif()
endif()
if(_viv_demo_smooth_enabled)
    message(STATUS "Examples: using SmoothScrollbar controllers")
else()
    message(STATUS "Examples: using native scrolling")
endif()

option(VIRTUALITEMVIEWS_EXAMPLES_USE_ASTER "Use optional aster ImageBox in image examples" ON)
add_library(viv_demo_images INTERFACE)
target_link_libraries(viv_demo_images INTERFACE Qt${QT_VERSION_MAJOR}::Widgets)
if(VIRTUALITEMVIEWS_EXAMPLES_USE_ASTER)
    # Inspect the package before importing it: a Qt6 package must not introduce
    # Qt6 targets into the Qt5 examples (and vice versa).
    find_file(_viv_aster_config NAMES asterConfig.cmake
        HINTS "${aster_DIR}" ENV aster_DIR ${CMAKE_PREFIX_PATH} ENV CMAKE_PREFIX_PATH
        PATH_SUFFIXES aster lib/cmake/aster)
    if(_viv_aster_config)
        file(STRINGS "${_viv_aster_config}" _viv_aster_qt
            REGEX "find_dependency\\(Qt[56]")
        if(_viv_aster_qt MATCHES "find_dependency\\(Qt${QT_VERSION_MAJOR}[^0-9]")
            get_filename_component(_viv_aster_dir "${_viv_aster_config}" DIRECTORY)
            find_package(aster CONFIG QUIET PATHS "${_viv_aster_dir}" NO_DEFAULT_PATH)
            if(aster_FOUND AND TARGET aster::gui)
                target_link_libraries(viv_demo_images INTERFACE aster::gui)
                target_compile_definitions(viv_demo_images INTERFACE VIV_DEMO_USE_ASTER=1)
                set(_viv_demo_aster_enabled TRUE)
            endif()
        else()
            message(STATUS "Image examples: aster package Qt version does not match; using threaded thumbnails")
        endif()
    endif()
    unset(_viv_aster_config CACHE)
endif()
if(_viv_demo_aster_enabled)
    message(STATUS "Image examples: using aster::gui ImageBox")
else()
    message(STATUS "Image examples: using threaded thumbnails")
endif()

    add_custom_target(viv_demo_scroll_tests DEPENDS tst_demoscrolling tst_demoscrolling_fallback)
    if(TARGET tst_smoothscrollbar)
        add_dependencies(viv_demo_scroll_tests tst_smoothscrollbar)
    endif()
endif()
